================
Counter actor
================

actor Counter do
  state count: 0

  on :increment do
    become count: count + 1
    reply :ok
  end

  on :decrement do
    become count: count - 1
    reply :ok
  end

  on :get do
    reply count
  end
end

================
Checkout with cons and dot access
================

actor Checkout do
  state items: [Item] :: [], total: Int :: 0

  on :add(item: Item) do
    become items: [item | items],
           total: total + item.price
    reply :ok
  end

  on :summary do
    reply {items, total}
  end
end
