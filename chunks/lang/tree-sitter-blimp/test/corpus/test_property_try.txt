================================================================================
Test and property blocks in an actor, and try with catch
================================================================================

actor Shapes do
  state n: Int :: 7
  on :kinds do
    reply n
  end
  test "seven kinds" do
    s = spawn Shapes
    assert_eq(s <- :kinds, 7)
  end
  property "n is never negative" do
    given k: gen_integer(0, 9)
    assert(k >= 0)
  end
end
r = try do
  1
catch e do
  e
end
